def Trefoil.Finite.ReducedForm : ℤ → ℤ → ℤ → Prop :=
fun h a s => 0 ≤ a ∧ a < h ∧ h ≤ s ∧ h % 2 = 0 ∧ s % 2 = 0 ∧ (2 * h - 1) * (2 * s - 1) - 4 * a ^ 2 = 245
def Trefoil.Finite.HasCorrectionComparison : ℤ → ℤ → ℤ → Prop :=
fun h a s =>
  ∃ label,
    (∀ (x : Trefoil.Finite.ZVector), label (x + x) = Trefoil.Finite.doubleLabel (label x)) ∧
      ∀ (x : Trefoil.Finite.ZVector) (z : Trefoil.Finite.QVector),
        ((Trefoil.Finite.surgeryMatrix h a s).mulVec z = fun i => ↑(x i)) →
          ∃ n, Trefoil.Finite.quadraticBound x z - Trefoil.Finite.dY (label x) = 2 * ↑n
theorem Trefoil.Consequences.lower_bound_245 : ∀ (u : ℕ),
  2 ≤ u → (u = 2 → ∃ h a s, Trefoil.Finite.ReducedForm h a s ∧ Trefoil.Finite.HasCorrectionComparison h a s) → 3 ≤ u :=
fun u hu realize =>
  Decidable.byContradiction fun hn =>
    have heq := Decidable.byContradiction fun a => Trefoil.Consequences.lower_bound_245._proof_1_1 u hu hn a;
    Exists.casesOn (realize heq) fun h h_1 =>
      Exists.casesOn h_1 fun a h_2 =>
        Exists.casesOn h_2 fun s h_3 =>
          And.casesOn h_3 fun hf hc => Trefoil.Finite.no_reduced_correction_comparison hf hc
theorem Trefoil.Consequences.lower_bound_lattice : ∀ (m u : ℕ),
  m = 4 ∨ 6 ≤ m → 2 ≤ u → (u = 2 → Nonempty (Trefoil.LatticeEmbedding.Embedding m)) → 3 ≤ u :=
fun m u hm hu realize =>
  Decidable.byContradiction fun hn =>
    have heq := Decidable.byContradiction fun a => Trefoil.Consequences.lower_bound_lattice._proof_1_1 u hu hn a;
    Nonempty.casesOn (realize heq) fun embedding => Trefoil.LatticeExtraction.no_embedding hm embedding
