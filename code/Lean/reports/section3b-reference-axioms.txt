def Trefoil.Section3BMinimum.ReducedAt : ℕ → ℤ → ℤ → ℤ → Prop :=
fun delta h a s => 0 ≤ a ∧ a < h ∧ h ≤ s ∧ h % 2 = 0 ∧ s % 2 = 0 ∧ (2 * h - 1) * (2 * s - 1) - 4 * a ^ 2 = ↑delta
def Trefoil.Section3BMinimum.spinBase : ℕ → ℕ → ℕ :=
fun p q => (p + (p - q) - 1) * ((p + 1) / 2) % p
def Trefoil.Section3BMinimum.lensCorrection : (p : ℕ) → ℕ → ZMod p → ℚ :=
fun p q j => Trefoil.Finite.correctionTerm p (p - q) ((Trefoil.Section3BMinimum.spinBase p q + j.val) % p)
